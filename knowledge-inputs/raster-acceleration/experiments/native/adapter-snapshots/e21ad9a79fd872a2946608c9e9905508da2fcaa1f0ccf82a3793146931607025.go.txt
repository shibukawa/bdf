// Pipeline experiments use only public APIs; separate renderers per worker.
package main

import (
	"bytes"
	"encoding/json"
	"flag"
	"image"
	"image/draw"
	"image/png"
	"os"
	"path/filepath"
	"runtime"
	"sync"
	"time"

	"github.com/shibukawa/bdf"
	"github.com/shibukawa/bdf/raster/imagebdf"
	"github.com/shibukawa/bdf/vector/pdfbdf"
)

func must(e error) {
	if e != nil {
		panic(e)
	}
}

var sampleCount = 5
var sampleDuration = 300 * time.Millisecond

func sample(f func()) []float64 {
	s := []float64{}
	for i := 0; i < sampleCount; i++ {
		t := time.Now()
		n := 0
		for time.Since(t) < sampleDuration {
			f()
			n++
		}
		s = append(s, float64(time.Since(t).Nanoseconds())/float64(n))
	}
	return s
}
func main() {
	input := flag.String("input", "", "BDF input")
	output := flag.String("output", "/tmp/bdf-pipeline", "output")
	dpi := flag.Float64("dpi", 72, "dpi")
	flag.IntVar(&sampleCount, "samples", 5, "number of samples")
	sampleMS := flag.Int("sample-ms", 300, "sample duration in ms")
	flag.Parse()
	sampleDuration = time.Duration(*sampleMS) * time.Millisecond
	must(os.MkdirAll(*output, 0755))
	rd, e := bdf.OpenSingleFile(*input)
	must(e)
	defer rd.Close()
	doc, e := rd.ToDocument()
	must(e)
	view := doc.Views[0]
	opts := &imagebdf.Options{NoSystemFonts: true, FontDirs: []string{"converter/pptx/testdata/fonts", "converter/docx/testdata/fonts", "fixture/testdata/fonts"}}
	newRenderer := func() *imagebdf.Renderer { return imagebdf.New(doc, opts) }
	r := newRenderer()
	t := time.Now()
	base, e := r.PageDPI(view, 0, *dpi)
	must(e)
	cold := time.Since(t).Nanoseconds()
	w, h := base.Bounds().Dx(), base.Bounds().Dy()
	result := map[string]any{"input": *input, "dpi": *dpi, "width": w, "height": h, "first_render_ns": cold, "cpu": runtime.GOMAXPROCS(0), "samples": sampleCount, "sample_ms": *sampleMS}
	var last *image.RGBA
	result["page_owned_ns"] = sample(func() { last, e = r.PageDPI(view, 0, *dpi); must(e) })
	result["page_to_png_ns"] = sample(func() {
		img, err := r.PageDPI(view, 0, *dpi)
		must(err)
		var out bytes.Buffer
		must(png.Encode(&out, img))
	})
	result["cached_page_copy_ns"] = sample(func() { last = image.NewRGBA(base.Bounds()); copy(last.Pix, base.Pix) })
	region := bdf.Rect{W: view.Pages[0].W / 2, H: view.Pages[0].H / 2}
	regionImage, e := r.Region(view, 0, region, max(w/2, 1), max(h/2, 1))
	must(e)
	result["region_owned_ns"] = sample(func() { last, e = r.Region(view, 0, region, max(w/2, 1), max(h/2, 1)); must(e) })
	result["cached_region_copy_ns"] = sample(func() { last = image.NewRGBA(regionImage.Bounds()); copy(last.Pix, regionImage.Pix) })
	object := bdf.NewObject()
	object.BBox = bdf.Rect{W: 128, H: 128}
	object.FillColor(0x2367abff).FillRect(.25, .5, 120, 120)
	objectImage, e := imagebdf.Object(object, *dpi/72, opts)
	must(e)
	result["object_owned_ns"] = sample(func() { last, e = imagebdf.Object(object, *dpi/72, opts); must(e) })
	result["cached_object_copy_ns"] = sample(func() { last = image.NewRGBA(objectImage.Bounds()); copy(last.Pix, objectImage.Pix) })
	workers := make([]*imagebdf.Renderer, 4)
	for i := range workers {
		workers[i] = newRenderer()
	}
	pageBatch := func(parallel bool) {
		var wg sync.WaitGroup
		for i := range workers {
			job := func(i int) { _, err := workers[i].PageDPI(view, 0, *dpi); must(err) }
			if parallel {
				wg.Add(1)
				go func() { defer wg.Done(); job(i) }()
			} else {
				job(i)
			}
		}
		wg.Wait()
	}
	result["four_pages_serial_ns"] = sample(func() { pageBatch(false) })
	result["four_pages_parallel_ns"] = sample(func() { pageBatch(true) })
	diffs := map[string]any{}
	for _, cfg := range []struct {
		name         string
		nx, ny, halo int
	}{{"bands-no-halo", 1, 4, 0}, {"bands-halo64", 1, 4, 64}, {"tiles-halo64", 2, 2, 64}, {"tiles-full-halo", 2, 2, max(w, h)}} {
		tile := func() {
			out := image.NewRGBA(base.Bounds())
			var wg sync.WaitGroup
			for i := 0; i < 4; i++ {
				wg.Add(1)
				go func(i int) {
					defer wg.Done()
					tx, ty := i%cfg.nx, i/cfg.nx
					rect := image.Rect(tx*w/cfg.nx, ty*h/cfg.ny, (tx+1)*w/cfg.nx, (ty+1)*h/cfg.ny)
					expanded := rect.Inset(-cfg.halo).Intersect(base.Bounds())
					sx, sy := float64(view.Pages[0].W)/float64(w), float64(view.Pages[0].H)/float64(h)
					region := bdf.Rect{X: float32(float64(expanded.Min.X) * sx), Y: float32(float64(expanded.Min.Y) * sy), W: float32(float64(expanded.Dx()) * sx), H: float32(float64(expanded.Dy()) * sy)}
					img, err := workers[i].Region(view, 0, region, expanded.Dx(), expanded.Dy())
					must(err)
					draw.Draw(out, rect, img, rect.Min.Sub(expanded.Min), draw.Src)
				}(i)
			}
			wg.Wait()
			last = out
		}
		result[cfg.name+"_ns"] = sample(tile)
		changed, maxError := 0, 0
		for i := 0; i < len(base.Pix); i += 4 {
			different := false
			for c := 0; c < 4; c++ {
				d := int(last.Pix[i+c]) - int(base.Pix[i+c])
				if d < 0 {
					d = -d
				}
				if d > maxError {
					maxError = d
				}
				different = different || d != 0
			}
			if different {
				changed++
			}
		}
		diffs[cfg.name] = map[string]int{"changed_pixels": changed, "max_channel_error": maxError}
		save(filepath.Join(*output, cfg.name+".png"), last)
	}
	result["region_quality"] = diffs
	var pdf bytes.Buffer
	pdfDoc, pdfView := *doc, *view
	pdfView.Pages = pdfView.Pages[:1]
	pdfDoc.Views = []*bdf.View{&pdfView}
	result["vector_pdf_ns"] = sample(func() {
		pdf.Reset()
		_, err := pdfbdf.Write(&pdf, &pdfDoc, &pdfbdf.Options{NoSystemFonts: true, FontDirs: opts.FontDirs})
		must(err)
	})
	result["vector_pdf_bytes"] = pdf.Len()
	must(os.WriteFile(filepath.Join(*output, "vector.pdf"), pdf.Bytes(), 0644))
	save(filepath.Join(*output, "page.png"), base)
	var encoded bytes.Buffer
	must(png.Encode(&encoded, base))
	result["raster_png_bytes"] = encoded.Len()
	data, e := json.MarshalIndent(result, "", "  ")
	must(e)
	must(os.WriteFile(filepath.Join(*output, "result.json"), append(data, '\n'), 0644))
	_ = last
}
func save(path string, img image.Image) {
	f, e := os.Create(path)
	must(e)
	must(png.Encode(f, img))
	must(f.Close())
}
