func TestExperimentSIMD(t *testing.T) {
	color := [4]float32{.12, .15, .25, .5}
	dst := make([]float32, 128)
	want := make([]float32, len(dst))
	for i := range dst {
		dst[i] = float32(i%11) / 12
		want[i] = color[i%4] + dst[i]*(1-color[3])
	}
	experimentSolidSIMD(&dst[0], len(dst)/4, &color)
	for i := range dst {
		if dst[i] != want[i] {
			t.Fatalf("SIMD float operation differs at %d: %.9g vs %.9g", i, dst[i], want[i])
		}
	}
}
func TestRasterExperimentOutputs(t *testing.T) {
	output := os.Getenv("BDF_EXPERIMENT_OUTPUT")
	if output == "" {
		t.Skip("experiment runner supplies output directory")
	}
	reference := os.Getenv("BDF_EXPERIMENT_REFERENCE")
	type row struct {
		Case             string  `json:"case"`
		SHA256           string  `json:"sha256"`
		Pixels           int     `json:"pixels"`
		ChangedPixels    int     `json:"changed_pixels"`
		MaxChannelError  int     `json:"max_channel_error"`
		MeanChannelError float64 `json:"mean_channel_error"`
		CacheBytes       int     `json:"cache_bytes"`
		CacheHits        int     `json:"cache_hits"`
		CacheMisses      int     `json:"cache_misses"`
	}
	rows := []row{}
	for _, name := range []string{"pdf/chrome-doc", "pptx/features", "drawio/showcase"} {
		doc := openDoc(t, "../../testdata/"+name+".bdf")
		view := doc.Views[0]
		for _, dpi := range []float64{72, 300} {
			for _, mode := range []struct {
				name string
				mode AntialiasMode
			}{{"on", AntialiasOn}, {"off", AntialiasOff}} {
				key := fmt.Sprintf("%s-%.0f-%s", filepath.Base(name), dpi, mode.name)
				r := New(doc, &Options{Antialias: mode.mode, FontDirs: testFonts, NoSystemFonts: true})
				first, err := r.PageDPI(view, 0, dpi)
				if err != nil {
					t.Fatal(err)
				}
				img, err := r.PageDPI(view, 0, dpi)
				if err != nil {
					t.Fatal(err)
				}
				assertImagesEqual(t, img, first)
				hash := sha256.Sum256(img.Pix)
				result := row{Case: key, SHA256: hex.EncodeToString(hash[:]), Pixels: img.Bounds().Dx() * img.Bounds().Dy()}
				if reference != "" {
					f, err := os.Open(filepath.Join(reference, key+".png"))
					if err != nil {
						t.Fatal(err)
					}
					ref, err := png.Decode(f)
					f.Close()
					if err != nil {
						t.Fatal(err)
					}
					sum := 0
					for y := 0; y < img.Bounds().Dy(); y++ {
						for x := 0; x < img.Bounds().Dx(); x++ {
							a := img.RGBAAt(x, y)
							b := ref.At(x, y)
							br, bg, bb, ba := b.RGBA()
							expected := [4]int{int(br >> 8), int(bg >> 8), int(bb >> 8), int(ba >> 8)}
							actual := [4]int{int(a.R), int(a.G), int(a.B), int(a.A)}
							changed := false
							for c := 0; c < 4; c++ {
								d := int(math.Abs(float64(actual[c] - expected[c])))
								sum += d
								result.MaxChannelError = max(result.MaxChannelError, d)
								changed = changed || d != 0
							}
							if changed {
								result.ChangedPixels++
							}
						}
					}
					result.MeanChannelError = float64(sum) / float64(result.Pixels*4)
				}
				if r.experiment != nil {
					result.CacheBytes = r.experiment.bytes
					result.CacheHits = r.experiment.hits
					result.CacheMisses = r.experiment.misses
				}
				f, err := os.Create(filepath.Join(output, key+".png"))
				if err != nil {
					t.Fatal(err)
				}
				if err := png.Encode(f, img); err != nil {
					t.Fatal(err)
				}
				f.Close()
				rows = append(rows, result)
			}
		}
	}
	data, err := json.MarshalIndent(rows, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(output, "quality.json"), append(data, '\n'), 0644); err != nil {
		t.Fatal(err)
	}
}

// Keep image imported for future heterogeneous outputs and enforce the
// current renderer's concrete return contract here.
var _ image.Image = (*image.RGBA)(nil)

// A scalar glyph mask loses subpixel occupancy and winding. Max-union cannot
// reproduce a run scan, even for two adjacent non-overlapping outlines.
func TestExperimentGlyphUnion(t *testing.T) {
	a, b := &path{}, &path{}
	a.rect(.1, 0, .5, 1)
	b.rect(.6, 0, .5, 1)
	bounds := image.Rect(0, 0, 2, 1)
	pa, pb := a.flatten(identity, flatTol), b.flatten(identity, flatTol)
	whole := rasterize(append(pa, pb...), bdf.NonZero, bounds)
	merged := union(rasterize(pa, bdf.NonZero, bounds), rasterize(pb, bdf.NonZero, bounds))
	if whole.at(0, 0) == merged.at(0, 0) {
		t.Fatal("counterexample unexpectedly identical")
	}
	t.Logf("combined glyph coverage=%g, max-union=%g", whole.at(0, 0), merged.at(0, 0))
}

func TestExperimentPrecision(t *testing.T) {
	rows := []map[string]any{}
	for _, mode := range []byte{bdf.BlendSourceOver, bdf.BlendMultiply, bdf.BlendScreen} {
		a := newSurface(8, 8)
		z := &surface{w: 8, h: 8, packed: make([]byte, 8*8*4)}
		cov := &mask{r: a.bounds()}
		a.fillBackground(cov, solid{1, 1, 1, 1})
		z.fillBackground(cov, solid{1, 1, 1, 1})
		for i := 0; i < 200; i++ {
			c := solid{.002, .004, .006, .01}
			a.fill(cov, c, 1, mode, cov)
			z.fill(cov, c, 1, mode, cov)
		}
		aa, zz := a.toRGBA(), z.toRGBA()
		delta := 0
		for i := range aa.Pix {
			d := int(aa.Pix[i]) - int(zz.Pix[i])
			if d < 0 {
				d = -d
			}
			delta = max(delta, d)
		}
		rows = append(rows, map[string]any{"blend": mode, "layers": 200, "max_channel_error": delta, "float_rgba": []int{int(aa.Pix[0]), int(aa.Pix[1]), int(aa.Pix[2]), int(aa.Pix[3])}, "packed_rgba": []int{int(zz.Pix[0]), int(zz.Pix[1]), int(zz.Pix[2]), int(zz.Pix[3])}})
	}
	if out := os.Getenv("BDF_EXPERIMENT_OUTPUT"); out != "" {
		data, e := json.MarshalIndent(rows, "", "  ")
		if e != nil {
			t.Fatal(e)
		}
		if e = os.WriteFile(filepath.Join(out, "precision.json"), append(data, '\n'), 0644); e != nil {
			t.Fatal(e)
		}
	}
}

func TestExperimentGeometry(t *testing.T) {
	cases := [][8]float64{{0, 0, 100, 0, -100, 0, 0, 0}, {0, 0, 0, 100, 100, -100, 100, 0}, {0, 0, 1, 1, 2, 2, 3, 3}, {0, 0, 1000, 1000, -1000, 1000, 0, 0}}
	maxDist := 0.
	for _, c := range cases {
		out := []point{{c[0], c[1]}}
		out = experimentAdaptive(out, point{c[0], c[1]}, point{c[2], c[3]}, point{c[4], c[5]}, point{c[6], c[7]}, flatTol, 0)
		for i := 0; i <= 1000; i++ {
			q := float64(i) / 1000
			u := 1 - q
			x := u*u*u*c[0] + 3*u*u*q*c[2] + 3*u*q*q*c[4] + q*q*q*c[6]
			y := u*u*u*c[1] + 3*u*u*q*c[3] + 3*u*q*q*c[5] + q*q*q*c[7]
			best := math.Inf(1)
			for j := 1; j < len(out); j++ {
				a, b := out[j-1], out[j]
				dx, dy := b.x-a.x, b.y-a.y
				v := 0.
				if dx*dx+dy*dy > 0 {
					v = min(max(((x-a.x)*dx+(y-a.y)*dy)/(dx*dx+dy*dy), 0), 1)
				}
				best = math.Min(best, math.Hypot(x-(a.x+v*dx), y-(a.y+v*dy)))
			}
			maxDist = math.Max(maxDist, best)
		}
	}
	if maxDist > flatTol {
		t.Fatalf("adaptive sampled distance %g > tolerance %g", maxDist, flatTol)
	}
	p := &path{}
	p.moveTo(0, 0)
	p.lineTo(10, 5)
	r := New(bdf.NewDocument(), nil)
	ls := lineStyle{width: 2, join: bdf.JoinMiter, miter: 10}
	bounds := image.Rect(0, 0, 32, 32)
	check := func() {
		a, _, _ := r.scratch().strokeMask(p, identity, ls, bounds)
		b, _, _ := (&scratch{}).strokeMask(p, identity, ls, bounds)
		for y := 0; y < 32; y++ {
			for x := 0; x < 32; x++ {
				av, bv := float32(0), float32(0)
				if image.Pt(x, y).In(a.r) {
					av = a.at(x, y)
				}
				if image.Pt(x, y).In(b.r) {
					bv = b.at(x, y)
				}
				if av != bv {
					t.Fatalf("stroke cache differs at %d,%d", x, y)
				}
			}
		}
	}
	check()
	check()
	ls.width = 4
	check()
	p.lineTo(20, 12)
	check()
}
